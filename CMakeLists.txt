cmake_minimum_required(VERSION 3.16)
project(cache_simulator VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(cache_core src/cache.cpp)
target_include_directories(cache_core PUBLIC include)
add_executable(cache_sim src/main.cpp)
target_link_libraries(cache_sim PRIVATE cache_core)
add_executable(cache_tests tests/cache_tests.cpp)
target_link_libraries(cache_tests PRIVATE cache_core)
foreach(target cache_core cache_sim cache_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
enable_testing()
add_test(NAME cache_core_tests COMMAND cache_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME cache_cli_tests COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:cache_sim>)
endif()
